const fetchTradingData = async () => { const strategyElement = document.getElementById("strategy"); const assetNameElement = document.getElementById("asset-name"); const asset = "SOLUSDT"; // Trading pair const stopLossPercent = 0.8; // Stop loss at 0.8% const target1Percent = 1.0; // Target 1: +1% const target2Percent = 2.0; // Target 2: +2% try { // Fetch data from Binance API const API_URL = `https://api.binance.com/api/v3/ticker/price?symbol=${asset}`; const response = await fetch(API_URL); if (!response.ok) { throw new Error(`API error, status code: ${response.status}`); } const priceData = await response.json(); if (!priceData.price) { throw new Error("API response does not contain a valid price."); } // Process price data const entryPrice = parseFloat(priceData.price); // Current price const stopLoss = entryPrice * (1 - stopLossPercent / 100); const target1 = entryPrice * (1 + target1Percent / 100); const target2 = entryPrice * (1 + target2Percent / 100); // Update strategy information assetNameElement.textContent = "SOL - Solana"; strategyElement.innerHTML = `

Strategy

Current Price: $${entryPrice.toFixed(2)}

Stop Loss: $${stopLoss.toFixed(2)} (-${stopLossPercent}%)

Profit Targets

Target 1: $${target1.toFixed(2)} (+${target1Percent}%)

Target 2: $${target2.toFixed(2)} (+${target2Percent}%)

`; } catch (error) { console.error("Error fetching trading data:", error); // Show error message strategyElement.innerHTML = `

Unable to load today's strategy. Check your internet connection or try again later.
Error: ${error.message}

`; } }; // Update the trading data on page load and every 5 minutes fetchTradingData(); setInterval(fetchTradingData, 300000); // Update every 5 minutes
